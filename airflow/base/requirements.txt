# Python packages baked into the custom Airflow image.
# These were previously injected at container startup via
# _PIP_ADDITIONAL_REQUIREMENTS in .env — pin/extend them here instead.
#
# Great Expectations 1.x is required by both consumers, and they must stay on
# the same major: dali/validation.py (context.data_sources, context.suites.add,
# gx.ValidationDefinition) and plugins/dataops/gx.py's validate_dataframe, which
# the vendored WaveStitchPlus checks go through. 1.x removed the 0.18 Validator
# API outright, so a downgrade silently breaks both.
great_expectations>=1.0,<2.0
pandas
apache-airflow-providers-amazon

# Required by the WaveStitchPlus DataOps pipeline that dali.processing runs
# (dataops.validation.pandera_schemas). The rest of that pipeline's imports —
# pandas, numpy, PyYAML, great_expectations — are already satisfied above or by
# Airflow itself; its heavy extras (torch, darts, pypots) belong to the
# imputation apps, which this pipeline never invokes.
# Pinned to WaveStitchPlus/requirements.txt's version; nothing else here uses
# pandera, so matching the pipeline that does is free.
pandera==0.20.4

# Imputation (dali.processing.impute_prepared_bundle), both engines of
# dataops.imputation_runner.
#
# scipy: the "pandas" engine reproduces Darts' interpolation filler in-process,
# and pandas hands every method except linear to SciPy — nearest, quadratic,
# cubic, slinear and zero fail without it.
#
# u8darts[notorch]: the "darts" engine runs the vendored Darts runner
# (plugins/Darts_app) in a subprocess, which is the only way to run darts/kalman.
# The runner needs MissingValuesFiller and KalmanFilter, neither of which needs
# torch, so the notorch variant is enough. It still brings about fifty packages
# (catboost, lightgbm, prophet, statsforecast, numba, matplotlib, ...), so this
# is the heavy line in the file. The image installs under Airflow's constraints
# file (see Dockerfile), which keeps numpy and scikit-learn at the versions
# Airflow was tested with; unconstrained, pip moves both.
scipy
u8darts[notorch]
